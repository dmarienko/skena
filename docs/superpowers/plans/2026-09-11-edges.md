```ts
import { GRID } from './constants';

export interface RouteNode { id: string; type: string; x: number; y: number; w: number; h: number; outputNodeId?: string }
export type Side = 'top' | 'right' | 'bottom' | 'left';
export interface RouteEdge { id: string; source: string; target: string; sourceSide?: Side; targetSide?: Side }
export type EdgeKind = 'sequence' | 'output' | 'context';
export interface RoutedEdge {
  id: string; kind: EdgeKind;
  points: [number, number][];      // - polyline, first = exit point on the source border, last = entry point on the target border
  variant: number;                 // - 0-based index among the edges sharing the source border (colour + exit slot)
  fallback: boolean;               // - true when no gap route was found (caller uses the old router)
}
export const LANE_STEP = 10;       // - px between parallel edges in one gap, and between exit points on one border
export const BEND_COST = GRID;     // - a corner costs one grid of length

export function edgeKind(e: RouteEdge, byId: Map<string, RouteNode>): EdgeKind {
  const s = byId.get(e.source), t = byId.get(e.target);
  if (!s || !t) return 'context';
  if (s.type === 'code' && s.outputNodeId === t.id) return 'output';
  if (s.type === 'code' && t.type === 'code') return 'sequence';
  return 'context';
}
```

- [ ] **Step 2: the gap grid** — a Hanan-style grid: candidate vertical lines at every node's `x − GRID/2` and `x + w + GRID/2` (the centre of the gap on each side), horizontal lines at `y − GRID/2` and `y + h + GRID/2`, plus the section's outer margin lines (min x − GRID/2, max right + GRID/2, same for y). Vertices = line crossings not inside any node box (a box is `x..x+w`, `y..y+h`); graph edges between consecutive vertices on a line when the segment between them crosses no node box. Complexity is fine for a section (≤ 100 nodes → ≤ 200 lines).

- [ ] **Step 3: exit points** — for every (node, side) collect the edges attached there (as source with that `sourceSide`, or as target with that `targetSide`; a missing side defaults from geometry: right when the other node is to the right, etc.). Sort them by the other end's position (y then x for left/right sides; x then y for top/bottom). Slot i of n gets offset `(i − (n − 1) / 2) · LANE_STEP` from the border's middle; the exit point is on the border line (`x + w` for right, etc.). `variant` = i for the edge's source border. A border holding a single edge takes the offset of the slot the other end gave that edge (0 when both borders hold one), so a facing pair runs straight across.

- [ ] **Step 4: routing** — for each edge (sorted by id for stability): project the exit point onto the nearest grid line perpendicular to the border (one step out of the node into the gap), same for the entry point; Dijkstra on the grid graph with cost = length + `BEND_COST` per direction change (state = vertex + incoming direction); the path's vertices → polyline `[exit, first grid point, …, last grid point, entry]`; merge collinear points. No path → `fallback: true`, `points: []`.

- [ ] **Step 5: lanes** — after all paths exist, walk each path's straight segments: a vertical segment on grid line x = c spanning y ∈ [a, b] occupies line c over [a, b]. For each line keep the intervals already taken with their lane index; a new segment takes the smallest lane index whose intervals do not overlap [a, b] (open intervals); lane k → offset from the line outwards, `0, −10, +10, −20 … ±40` (k mod 9), so a lone run keeps the gap centre line and a pair straddles it. Shift the segment's x by the offset; connect neighbours by keeping the corner at the shifted coordinates (a corner between a vertical segment at x' and a horizontal segment at y' is (x', y')). Same for horizontal segments. Exit/entry segments (the first and last step out of a node) are not lane-shifted.
